    -- PASSO 1: DROP LIMPO

    DROP TABLE IF EXISTS tbl_filhos;
    DROP TABLE IF EXISTS tbl_conjuges;
    DROP TABLE IF EXISTS tbl_avisos;
    DROP TABLE IF EXISTS tbl_usuarios;
    DROP DATABASE IF EXISTS sistema_familia;

    -- PASSO 2: CRIAR BANCO
    -- ============================================================================
    CREATE DATABASE sistema_familia CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci;
    USE sistema_familia;

    -- PASSO 3: TABELA DE USUÁRIOS
    -- ============================================================================
    CREATE TABLE tbl_usuarios (
        id_usuario BIGINT AUTO_INCREMENT PRIMARY KEY,
        no_usuario VARCHAR(120) NOT NULL UNIQUE,
        pw_senha VARCHAR(255) NOT NULL,
        dt_nascimento DATE,
        ds_email VARCHAR(120),
        nu_vertical INT,
        nu_horizontal INT,
        nu_whatsapp VARCHAR(20),
        lk_foto VARCHAR(255),
        role VARCHAR(20) DEFAULT 'user',
        criado_em DATETIME DEFAULT CURRENT_TIMESTAMP,
        atualizado_em DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
        deletado_em DATETIME NULL,
        
        INDEX idx_email (ds_email),
        INDEX idx_role (role),
        INDEX idx_nome (no_usuario)
    );

    -- PASSO 4: TABELA DE CÔNJUGES
    -- ============================================================================
    CREATE TABLE tbl_conjuges (
        id_conjuge BIGINT AUTO_INCREMENT PRIMARY KEY,
        usuario_a BIGINT NOT NULL,
        usuario_b BIGINT NOT NULL,
        dt_casamento DATE,
        ds_local VARCHAR(120),
        fl_divorcio BOOLEAN DEFAULT FALSE,
        dt_fim_casamento DATE NULL,
        criado_em DATETIME DEFAULT CURRENT_TIMESTAMP,
        atualizado_em DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
        deletado_em DATETIME NULL,
        
        FOREIGN KEY (usuario_a) REFERENCES tbl_usuarios(id_usuario),
        FOREIGN KEY (usuario_b) REFERENCES tbl_usuarios(id_usuario),
        
        INDEX idx_usuario_a (usuario_a),
        INDEX idx_usuario_b (usuario_b),
        INDEX idx_casal (usuario_a, usuario_b),
        INDEX idx_deletado (deletado_em)
    );

    -- PASSO 5: TABELA DE FILHOS
    -- ============================================================================
    CREATE TABLE tbl_filhos (
        id_filho BIGINT AUTO_INCREMENT PRIMARY KEY,
        id_usuario_filho BIGINT NOT NULL,
        id_pai BIGINT,
        id_mae BIGINT,
        criado_em DATETIME DEFAULT CURRENT_TIMESTAMP,
        atualizado_em DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
        deletado_em DATETIME NULL,
        
        FOREIGN KEY (id_usuario_filho) REFERENCES tbl_usuarios(id_usuario),
        FOREIGN KEY (id_pai) REFERENCES tbl_usuarios(id_usuario),
        FOREIGN KEY (id_mae) REFERENCES tbl_usuarios(id_usuario),
        
        INDEX idx_filho (id_usuario_filho),
        INDEX idx_pai (id_pai),
        INDEX idx_mae (id_mae),
        INDEX idx_genitor (id_pai, id_mae),
        INDEX idx_deletado (deletado_em)
    );

    -- PASSO 6: TABELA DE AVISOS
    -- ============================================================================
    CREATE TABLE tbl_avisos (
        id_aviso BIGINT AUTO_INCREMENT PRIMARY KEY,
        titulo VARCHAR(255) NOT NULL,
        mensagem TEXT NOT NULL,
        data_criacao DATETIME DEFAULT CURRENT_TIMESTAMP,
        
        INDEX idx_data (data_criacao)
    );

    -- PASSO 15: VERIFICAÇÃO FINAL

    SELECT '=== USUÁRIOS ===' AS info;
    SELECT id_usuario, no_usuario, role, dt_nascimento FROM tbl_usuarios WHERE deletado_em IS NULL ORDER BY id_usuario;

    SELECT '=== CÔNJUGES ===' AS info;
    SELECT 
    c.id_conjuge, 
    u1.no_usuario AS 'Cônjuge A', 
    u2.no_usuario AS 'Cônjuge B', 
    c.dt_casamento, 
    c.fl_divorcio 
    FROM tbl_conjuges c
    LEFT JOIN tbl_usuarios u1 ON c.usuario_a = u1.id_usuario
    LEFT JOIN tbl_usuarios u2 ON c.usuario_b = u2.id_usuario
    WHERE c.deletado_em IS NULL
    ORDER BY c.id_conjuge;

    SELECT '=== FILHOS ===' AS info;
    SELECT 
    f.id_filho, 
    uf.no_usuario AS 'Filho', 
    up.no_usuario AS 'Pai', 
    um.no_usuario AS 'Mãe'
    FROM tbl_filhos f
    JOIN tbl_usuarios uf ON f.id_usuario_filho = uf.id_usuario
    LEFT JOIN tbl_usuarios up ON f.id_pai = up.id_usuario
    LEFT JOIN tbl_usuarios um ON f.id_mae = um.id_usuario
    WHERE f.deletado_em IS NULL
    ORDER BY f.id_filho;

    SELECT '=== CONTAGEM ===' AS info;
    SELECT 
    COUNT(DISTINCT id_usuario) AS total_usuarios,
    (SELECT COUNT(*) FROM tbl_conjuges WHERE deletado_em IS NULL) AS total_casamentos,
    (SELECT COUNT(*) FROM tbl_filhos WHERE deletado_em IS NULL) AS total_filhos
    FROM tbl_usuarios 
    WHERE deletado_em IS NULL;
